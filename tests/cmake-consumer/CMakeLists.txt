cmake_minimum_required(VERSION 3.24)
project(sequel_consumer LANGUAGES C CXX)
find_package(sequel CONFIG REQUIRED)
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE sequel::sequel)
enable_testing()
add_test(NAME installed-sequel COMMAND consumer)
if(TARGET sequel::cxx)
  add_executable(consumer-cxx ../cxx_api_test.cpp)
  target_link_libraries(consumer-cxx PRIVATE sequel::cxx)
  add_test(NAME installed-sequel-cxx COMMAND consumer-cxx)
endif()
if(WIN32)
  set_property(TEST installed-sequel APPEND PROPERTY ENVIRONMENT_MODIFICATION
    "PATH=path_list_prepend:$<TARGET_FILE_DIR:sequel::sequel>")
  if(TARGET sequel::cxx)
    set_property(TEST installed-sequel-cxx APPEND PROPERTY ENVIRONMENT_MODIFICATION
      "PATH=path_list_prepend:$<TARGET_FILE_DIR:sequel::sequel>")
  endif()
endif()
